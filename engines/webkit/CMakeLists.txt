cmake_minimum_required(VERSION 3.20)
project(MochiOSWebKitABI LANGUAGES C CXX)
include(CTest)
option(MWK_BUILD_ENGINE "Build a real engine, never a stub shared library" OFF)
set(MWK_BACKEND_SOURCES "" CACHE STRING "Reviewed native WPE backend source files")
if(MWK_BUILD_ENGINE)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "mochiOS")
        message(FATAL_ERROR "libwebkit.so is a mochiOS product; host reference builds are not distributable")
    endif()
    if(NOT CMAKE_CROSSCOMPILING OR NOT CMAKE_SYSROOT OR NOT IS_DIRECTORY "${CMAKE_SYSROOT}")
        message(FATAL_ERROR "A real mochiOS cross SDK and sysroot are required")
    endif()
    foreach(mode LIBRARY INCLUDE PACKAGE)
        if(NOT CMAKE_FIND_ROOT_PATH_MODE_${mode} STREQUAL "ONLY")
            message(FATAL_ERROR "Target ${mode} searches must be isolated")
        endif()
    endforeach()
    if(NOT CMAKE_FIND_ROOT_PATH_MODE_PROGRAM STREQUAL "NEVER")
        message(FATAL_ERROR "Generators must be host programs")
    endif()
    include(CheckCSourceCompiles)
    check_c_source_compiles("#if !defined(__mochios__) || defined(__linux__)\n#error Wrong target compiler\n#endif\nint main(void) { return 0; }" MWK_REAL_SDK)
    if(NOT MWK_REAL_SDK)
        message(FATAL_ERROR "Compiler must target mochiOS; renaming host output is forbidden")
    endif()
    if(NOT MWK_BACKEND_SOURCES)
        message(FATAL_ERROR "No native WebKit backend: refusing to produce an empty libwebkit.so")
    endif()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(WPE REQUIRED IMPORTED_TARGET wpe-webkit-2.0)
    pkg_check_modules(WPE_HEADLESS REQUIRED IMPORTED_TARGET wpe-platform-headless-2.0)
    add_library(webkit SHARED src/abi.c ${MWK_BACKEND_SOURCES})
    target_include_directories(webkit PUBLIC include)
    target_link_libraries(webkit PRIVATE PkgConfig::WPE PkgConfig::WPE_HEADLESS)
    target_compile_features(webkit PRIVATE cxx_std_20)
    # Pass explicit target-prefix paths to both the linker and SDK import checks.
    target_link_directories(webkit PRIVATE ${WPE_LIBRARY_DIRS} ${WPE_HEADLESS_LIBRARY_DIRS})
    set_target_properties(webkit PROPERTIES VERSION 1.0.0 SOVERSION 1
        C_VISIBILITY_PRESET hidden CXX_VISIBILITY_PRESET hidden)
    target_link_options(webkit PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
    set_property(TARGET webkit APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
    install(TARGETS webkit LIBRARY DESTINATION lib)
    install(FILES include/mochios/webkit.h DESTINATION include/mochios)
endif()
if(BUILD_TESTING)
    add_executable(webkit_abi_contract tests/contract.c src/abi.c)
    target_include_directories(webkit_abi_contract PRIVATE include)
    target_compile_features(webkit_abi_contract PRIVATE c_std_11)
    add_test(NAME webkit_abi_contract COMMAND webkit_abi_contract)
    add_library(webkit_cpp_contract OBJECT tests/contract.cpp)
    target_include_directories(webkit_cpp_contract PRIVATE include)
    target_compile_features(webkit_cpp_contract PRIVATE cxx_std_23)
endif()
